# 수납 장부 V4 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 엑셀 일일장부를 앱 화면 「오늘 장부」로 옮기고, 수납 한 줄에서 처방·해피콜·약장 나감이 자동으로 생기게 한다.

**Architecture:** 순수 규칙(`lib/ledgerRules.ts`: 약어 파싱·합계)과 데이터 접근(`lib/ledger.ts`)을 나누고, 화면은 `app/ledger/page.tsx` + 컴포넌트 3개. 표 5개(`ledger_codes`·`ledger_days`·`ledger_entries`·`ledger_items`·`ledger_expenses`)를 새로 만들고 마감된 날은 RLS로 잠근다. 한 줄 저장은 클라이언트에서 순서대로(줄 → 처방·해피콜 → 약장 나감) 넣고 실패하면 앞 단계를 되돌린다.

**Tech Stack:** Next.js 16 App Router, React 19, Tailwind 4, supabase-js 2 (anon + RLS), Vitest, exceljs(엑셀 내려받기).

**Spec:** `docs/superpowers/specs/2026-10-02-수납장부-v4-design.md`

## Global Constraints

- 화면 글은 접수실이 읽는 한국어 존댓말, 개발 용어 금지. 버튼은 행동 그대로.
- 환자 연락처는 목록에서 가린다(`maskPhone`).
- 장부 줄·지출은 마감 전엔 고치고 지울 수 있고, 마감 뒤엔 정정 줄만. 약장 나감은 지우지 못하므로 줄을 지우면 되돌림 입고(`in`)를 넣는다.
- 탕약 항목 → `planHappyCalls` 그대로. 수령일 기본 = 그 날짜(장부 날짜). 포 수 없으면 처방을 만들지 않고 줄에 표시만.
- 확인 항목 기본값: 매출은 현황에서 접어 두기만(비밀번호 없음), 10월부터, 마감+정정 줄, 약어 표 초안 그대로 넣고 앱에서 고침, 수령일 기본 오늘, 환자 연락처는 선택.
- 테스트 `npx vitest run` · `npx eslint .` · `npx tsc --noEmit` 통과 뒤 커밋(Conventional Commits, 한국어).
- `.env.local`·키 커밋 금지. 실제 환자로 시험하지 않는다. 확인용 가짜 환자는 끝나면 지운다.

import { describe, it, expect } from "vitest";
import { parseNote, sumEntries, nextSeq, won } from "./ledgerRules";
import type { LedgerCode } from "./types";

const c = (code: string, name: string, group: LedgerCode["group"], extra: Partial<LedgerCode> = {}): LedgerCode => ({
  id: 0, code, name, group, cabinet_item_id: null, decoction_kind: null, active: true, sort_order: 0, created_at: "", ...extra,
});
const CODES: LedgerCode[] = [
  c("전", "전침", "treatment"), c("습", "습부항", "treatment"), c("V", "약침", "treatment"), c("mo", "mo", "treatment"),
  c("보험(처방)", "보험 탕약", "decoction", { decoction_kind: "insurance" }),
  c("일반", "일반 탕약", "decoction", { decoction_kind: "general" }),
  c("발효", "발효 탕약", "decoction", { decoction_kind: "fermented" }),
  c("공진단", "공진단", "cabinet", { cabinet_item_id: 1 }), c("소합원", "소합원", "cabinet", { cabinet_item_id: 4 }),
  c("오적", "오적산", "extract"), c("향사평위", "향사평위산", "extract"),
];

describe("parseNote", () => {
  it("쉼표로 나누고 아는 약어는 항목으로", () => {
    const r = parseNote("전,습,mo-0.6,향사평위2일", CODES);
    expect(r.items.map((i) => [i.name, i.group, i.qty, i.days])).toEqual([
      ["전침", "treatment", 1, null], ["습부항", "treatment", 1, null], ["mo", "treatment", 1, null], ["향사평위산", "extract", 1, 2],
    ]);
    expect(r.items.every((i) => !i.unknown)).toBe(true);
  });
  it("괄호 숫자는 만원 금액, #N은 분할 횟수", () => {
    const [i] = parseNote("일반(42.#2)", CODES).items;
    expect(i).toMatchObject({ name: "일반 탕약", group: "decoction", amount: 420000, split: 2, decoction_kind: "general" });
    expect(parseNote("발효(48)", CODES).items[0]).toMatchObject({ amount: 480000, split: null });
  });
  it("보험(처방)은 괄호째 약어", () => {
    expect(parseNote("V,보험(처방)", CODES).items[1]).toMatchObject({ name: "보험 탕약", decoction_kind: "insurance", amount: null });
  });
  it("숫자+T/알/개는 수량, 약장 품목 연결", () => {
    expect(parseNote("공진단5T", CODES).items[0]).toMatchObject({ qty: 5, cabinet_item_id: 1 });
    expect(parseNote("소합원 3알", CODES).items[0]).toMatchObject({ qty: 3 });
    expect(parseNote("V2", CODES).items[0]).toMatchObject({ name: "약침", qty: 2 });
  });
  it("리뷰 조각은 항목이 아니라 표시", () => {
    const r = parseNote("소합원3T,리뷰", CODES);
    expect(r.review).toBe(true);
    expect(r.items).toHaveLength(1);
  });
  it("모르는 말은 unknown으로 남긴다", () => {
    const r = parseNote("전,습부2", CODES);
    expect(r.items[1]).toMatchObject({ raw: "습부2", unknown: true });
  });
  it("대소문자·공백 무시, 빈 조각 무시", () => {
    expect(parseNote(" v , ,전 ", CODES).items.map((i) => i.name)).toEqual(["약침", "전침"]);
    expect(parseNote("", CODES).items).toEqual([]);
  });
});

describe("합계·번호", () => {

    expect(sumEntries([{ cash: 1000, cash_receipt: 0, card: 500 }, { cash: 0, cash_receipt: 200, card: 0 }])).toEqual({ cash: 1000, cash_receipt: 200, card: 500, subtotal: 1700 });
  });
  it("nextSeq는 가장 큰 번호 + 1, 비면 1", () => {
    expect(nextSeq([])).toBe(1);
    expect(nextSeq([{ seq: 3 }, { seq: 7 }])).toBe(8);
  });
  it("won", () => expect(won(1409500)).toBe("1,409,500"));
});

  decoction: "탕약", cabinet: "약장 품목", treatment: "치료", extract: "엑스제", other: "기타",

export type ParsedItem = {
  raw: string; code: string; name: string; group: LedgerGroup; qty: number;
  amount: number | null; days: number | null; split: number | null;
  cabinet_item_id: number | null; decoction_kind: DecoctionKind | null; unknown: boolean;
};

const norm = (s: string) => s.replace(/\s+/g, "").toLowerCase();

/** 기타 칸 글 → 항목. 리뷰 조각은 항목이 아니라 표시(review)로 돌려준다. */
export function parseNote(raw: string, codes: LedgerCode[]): { items: ParsedItem[]; review: boolean } {
  const active = codes.filter((c) => c.active);
  const byCode = new Map(active.map((c) => [norm(c.code), c]));
  const items: ParsedItem[] = [];
  let review = false;
  for (const piece of raw.split(",")) {
    const token = piece.trim();
    if (!token) continue;
    const t = norm(token);
    if (t === "리뷰" || t === "리뷰증정") { review = true; continue; }
    const whole = byCode.get(t);
    if (whole) { items.push(make(token, whole, { qty: 1 })); continue; }

    let rest = t;
    let days: number | null = null, amount: number | null = null, split: number | null = null, qty = 1;
    let m = rest.match(/(\d+)일$/);
    if (m) { days = Number(m[1]); rest = rest.slice(0, -m[0].length); }
    m = rest.match(/\((\d+)(?:\.#(\d+))?\)$/);
    if (m) { amount = Number(m[1]) * 10000; if (m[2]) split = Number(m[2]); rest = rest.slice(0, -m[0].length); }
    m = rest.match(/#(\d+)$/);
    if (m) { split = Number(m[1]); rest = rest.slice(0, -m[0].length); }
    m = rest.match(/(\d+)(t|알|개|p)?$/);
    if (m && rest.length > m[0].length) { qty = Number(m[1]); rest = rest.slice(0, -m[0].length); }
    m = rest.match(/-[\d.]+(\*\d+)?$/);
    if (m) rest = rest.slice(0, -m[0].length);
    const found = byCode.get(rest);
    if (!found) {
      items.push({ raw: token, code: token, name: token, group: "other", qty: 1, amount: null, days: null, split: null, cabinet_item_id: null, decoction_kind: null, unknown: true });
      continue;
    }
    items.push(make(token, found, { qty, amount, days, split }));
  }
  return { items, review };
}

function make(raw: string, c: LedgerCode, p: { qty: number; amount?: number | null; days?: number | null; split?: number | null }): ParsedItem {
  return {
    raw, code: c.code, name: c.name, group: c.group, qty: p.qty,
    amount: p.amount ?? null, days: p.days ?? null, split: p.split ?? null,
    cabinet_item_id: c.cabinet_item_id, decoction_kind: c.decoction_kind, unknown: false,
  };
}

export function sumEntries(entries: { cash: number; cash_receipt: number; card: number }[]) {
  const t = { cash: 0, cash_receipt: 0, card: 0, subtotal: 0 };
  for (const e of entries) { t.cash += e.cash; t.cash_receipt += e.cash_receipt; t.card += e.card; }
  t.subtotal = t.cash + t.cash_receipt + t.card;
  return t;
}
export function sumExpenses(expenses: { amount: number }[]): number {
  return expenses.reduce((s, e) => s + e.amount, 0);
}
export function nextSeq(entries: { seq: number }[]): number {
  return entries.reduce((m, e) => Math.max(m, e.seq), 0) + 1;
}
export function won(n: number): string { return n.toLocaleString("ko-KR"); }
/** "11,000" · "11000" · "1.1만" 같은 입력을 원 단위 정수로. 빈 값은 0. */
export function parseWon(s: string): number {
  const t = s.replace(/[,\s원]/g, "");
  if (!t) return 0;
  const m = t.match(/^(\d+(?:\.\d+)?)만$/);
  if (m) return Math.round(Number(m[1]) * 10000);
  const n = Number(t);
  return Number.isFinite(n) ? Math.round(n) : 0;
}
